def caesar(text, key):
    result = ""

    for c in text:
        if c.isalpha():
            base = ord('A') if c.isupper() else ord('a')
            result += chr((ord(c) - base + key) % 26 + base)
        else:
            result += c

    return result

text = input("Enter plaintext: ")
key = int(input("Enter key: "))

cipher = caesar(text, key)
plain = caesar(cipher, -key)

print("Encrypted:", cipher)
print("Decrypted:", plain)



------------------------------------playfair cipher -----------------------------------------


def matrix(key):
    s = ""
    for c in key.upper() + "ABCDEFGHIKLMNOPQRSTUVWXYZ":
        c = "I" if c == "J" else c
        if c.isalpha() and c not in s:
            s += c

    return [s[i:i+5] for i in range(0, 25, 5)]


def prepare(text):
    text = "".join(c for c in text.upper() if c.isalpha()).replace("J", "I")
    result = ""
    i = 0

    while i < len(text):
        a = text[i]

        if i + 1 == len(text):
            result += a + "X"
            i += 1
        elif a == text[i + 1]:
            result += a + "X"
            i += 1
        else:
            result += a + text[i + 1]
            i += 2

    return result


def playfair(text, key, decrypt=False):
    m = matrix(key)
    pos = {m[r][c]: (r, c) for r in range(5) for c in range(5)}

    if not decrypt:
        text = prepare(text)

    result = ""
    shift = -1 if decrypt else 1

    for i in range(0, len(text), 2):
        a, b = text[i], text[i + 1]
        r1, c1 = pos[a]
        r2, c2 = pos[b]

        if r1 == r2:
            result += m[r1][(c1 + shift) % 5]
            result += m[r2][(c2 + shift) % 5]

        elif c1 == c2:
            result += m[(r1 + shift) % 5][c1]
            result += m[(r2 + shift) % 5][c2]

        else:
            result += m[r1][c2] + m[r2][c1]

    return result


key = input("Enter key: ")
text = input("Enter plaintext: ")

cipher = playfair(text, key)
print("Encrypted:", cipher)
print("Decrypted:", playfair(cipher, key, True))


-----------------------------Hill chipher ----------------------------------------

def hill(text, key):
    text = "".join(c for c in text.upper() if c.isalpha())

    while len(text) % 3:
        text += "X"

    result = ""

    for i in range(0, len(text), 3):
        block = [ord(c) - 65 for c in text[i:i+3]]

        for row in key:
            value = sum(row[j] * block[j] for j in range(3))
            result += chr(value % 26 + 65)

    return result


key = [
    [6, 24, 1],
    [13, 16, 10],
    [20, 17, 15]
]

text = input("Enter plaintext: ")
cipher = hill(text, key)

print("Encrypted:", cipher)



-------------------------------Rail_fence----------------------------
def encrypt(text, key):
    rails = [[] for _ in range(key)]

    row = 0
    step = 1

    for ch in text:
        rails[row].append(ch)

        if row == 0:
            step = 1
        elif row == key - 1:
            step = -1

        row += step

    return "".join("".join(rail) for rail in rails)


def decrypt(cipher, key):
    pattern = []
    row = 0
    step = 1

    for _ in cipher:
        pattern.append(row)

        if row == 0:
            step = 1
        elif row == key - 1:
            step = -1

        row += step

    counts = [pattern.count(i) for i in range(key)]

    rails = []
    index = 0

    for count in counts:
        rails.append(list(cipher[index:index + count]))
        index += count

    result = ""

    for row in pattern:
        result += rails[row].pop(0)

    return result


text = input("Enter plaintext: ")
key = int(input("Enter number of rails: "))

cipher = encrypt(text, key)
print("Encrypted:", cipher)
print("Decrypted:", decrypt(cipher, key))

import math

def get_order(key):
    return sorted(range(len(key)), key=lambda k: key[k])


def encrypt(text, key):
    cols = len(key)
    rows = math.ceil(len(text) / cols)

    text += 'X' * (rows * cols - len(text))

    matrix = [list(text[i:i + cols])
              for i in range(0, len(text), cols)]

    order = get_order(key)

    cipher = ""

    for c in order:
        for r in range(rows):
            cipher += matrix[r][c]

    return cipher


def decrypt(cipher, key):
    cols = len(key)
    rows = math.ceil(len(cipher) / cols)

    order = get_order(key)

    matrix = [['' for _ in range(cols)] for _ in range(rows)]

    index = 0

    for c in order:
        for r in range(rows):
            matrix[r][c] = cipher[index]
            index += 1

    plain = ""

    for r in matrix:
        plain += "".join(r)

    return plain.rstrip('X')


text = input("Enter plaintext: ")
key = input("Enter key: ")

cipher = encrypt(text, key)
print("Encrypted:", cipher)

plain = decrypt(cipher, key)
print("Decrypted:", plain)

from Crypto.Cipher import DES
from Crypto.Util.Padding import pad, unpad 

key = b"12345678"

cipher = DES.new(key, DES.MODE_ECB)

plaintext = "Hello DES"

encrypted = cipher.encrypt(pad(plaintext.encode(), DES.block_size))

decipher = DES.new(key, DES.MODE_ECB)

decrypted = unpad(decipher.decrypt(encrypted), DES.block_size)

print("Encrypted:", encrypted.hex())
print("Decrypted:", decrypted.decode())

def power(a, b, p):
    return pow(a, b, p)

print("Bob's private key:", b)

======================== Blowfish ========================

from Crypto.Cipher import Blowfish
from Crypto.Util.Padding import pad, unpad
import os

# Generate a random key of 32 bytes
key = os.urandom(32)  # Key length: 32 to 448 bits
cipher = Blowfish.new(key, Blowfish.MODE_CBC)

# Data to encrypt
data = b"This is a secret message."

# Pad the data to ensure it's a multiple of 8 bytes
padded_data = pad(data, Blowfish.block_size)

# Encrypt the data
iv = cipher.iv  # Initialization vector
encrypted_data = cipher.encrypt(padded_data)
print("Encrypted:", encrypted_data)
cipher_decrypt = Blowfish.new(key, Blowfish.MODE_CBC, iv)

# Decrypt the data
decrypted_data = unpad(cipher_decrypt.decrypt(encrypted_data), Blowfish.block_size)
print("Decrypted:", decrypted_data)




======================== Digital Signature Standard (DSS) ========================

from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import dsa

private_key = dsa.generate_private_key(key_size=2048)
public_key = private_key.public_key()

message = b"Secure document transmission data."
signature = private_key.sign(message, hashes.SHA256())

print("Signature generated. Length:", len(signature))
 
try:
    public_key.verify(signature, message, hashes.SHA256())
    print("Verification SUCCESS: Signature is valid.")
except Exception as e:
    print("Verification FAILED:", e)




======================== Secure Socket Layer (SSL) ========================

Server.py-

import socket, ssl

s = socket.socket()
s.bind(('localhost', 12345))
s.listen(1)

ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
ctx.load_cert_chain('cert.pem', 'key.pem')

c, _ = s.accept()
c = ctx.wrap_socket(c, server_side=True)

msg = c.recv(1024).decode()
print("Client:", msg)

c.send(b"Hello Client! Secure message received.")
c.close()
s.close()


Client.py-

import socket, ssl

ctx = ssl.create_default_context()
ctx.check_hostname = False
ctx.verify_mode = ssl.CERT_NONE

s = socket.create_connection(('localhost', 12345))
c = ctx.wrap_socket(s, server_hostname='localhost')

c.send(b"Hello Server! This is a secure message.")
print("Server:", c.recv(1024).decode())

c.close()
